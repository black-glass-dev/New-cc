cmake_minimum_required(VERSION 3.20)
project(OwnColorEngine VERSION 0.1.0 LANGUAGES C CXX)
option(COLOR_BUILD_TESTS "Build standalone color reference tests" ON)
option(COLOR_SANITIZE "Address/undefined sanitizers for host tests" OFF)
option(COLOR_REQUIRE_SHADER_VALIDATION "Fail configure when GLSL compiler was not supplied" OFF)
set(COLOR_GLSLANG "" CACHE FILEPATH "Path to glslang/glslangValidator for SPIR-V compilation")
find_package(Threads REQUIRED)
if(COLOR_REQUIRE_SHADER_VALIDATION AND NOT COLOR_GLSLANG)
 message(FATAL_ERROR "Set COLOR_GLSLANG to a real glslang compiler")
endif()
add_library(own_color_engine src/science.cpp src/operators.cpp src/lut.cpp src/pipeline.cpp src/analysis_tools.cpp src/spatial.cpp src/tracking.cpp src/archive.cpp src/mastering.cpp src/gpu_graph.cpp bridge/color_c_api.cpp)
add_library(color::engine ALIAS own_color_engine)
target_link_libraries(own_color_engine PUBLIC Threads::Threads)
target_include_directories(own_color_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/bridge)
target_compile_features(own_color_engine PUBLIC cxx_std_20)
set_target_properties(own_color_engine PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(own_color_engine PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
 if(COLOR_SANITIZE)
  target_compile_options(own_color_engine PUBLIC -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
  target_link_options(own_color_engine PUBLIC -fsanitize=address,undefined,float-cast-overflow)
 endif()
endif()
if(COLOR_BUILD_TESTS)
 enable_testing()
 add_executable(color_tests tests/color_tests.cpp)
 target_link_libraries(color_tests PRIVATE color::engine)
 if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(color_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
 endif()
 add_test(NAME color_reference COMMAND color_tests)
 add_executable(color_extended_tests tests/extended_tests.cpp)
 target_link_libraries(color_extended_tests PRIVATE color::engine)
 target_compile_definitions(color_extended_tests PRIVATE COLOR_TEST_OUTPUT_DIR="${CMAKE_CURRENT_BINARY_DIR}/test_artifacts")
 if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(color_extended_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
 endif()
 add_test(NAME color_extended COMMAND color_extended_tests)
 add_executable(color_c_abi_test tests/c_abi_test.c)
 target_link_libraries(color_c_abi_test PRIVATE color::engine)
 set_target_properties(color_c_abi_test PROPERTIES LINKER_LANGUAGE CXX)
 add_test(NAME color_c_abi COMMAND color_c_abi_test)
 add_executable(color_bench tests/color_bench.cpp)
 target_link_libraries(color_bench PRIVATE color::engine)
endif()
if(COLOR_GLSLANG)
 file(GLOB COLOR_SHADER_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/common/*.inc")
 file(GLOB COLOR_GLSL_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/glsl/*.comp")
 foreach(SHADER IN LISTS COLOR_GLSL_SOURCES)
  get_filename_component(SHADER_NAME "${SHADER}" NAME)
  set(SPV "${CMAKE_CURRENT_BINARY_DIR}/spirv/${SHADER_NAME}.spv")
  add_custom_command(OUTPUT "${SPV}" COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/spirv"
   COMMAND "${COLOR_GLSLANG}" -V --target-env vulkan1.1 "${SHADER}" -o "${SPV}" DEPENDS "${SHADER}" ${COLOR_SHADER_HEADERS} VERBATIM)
  list(APPEND COLOR_SPIRV_OUTPUTS "${SPV}")
  if(COLOR_BUILD_TESTS)
   add_test(NAME shader_${SHADER_NAME} COMMAND "${COLOR_GLSLANG}" -V --target-env vulkan1.1 "${SHADER}" -o "${SPV}")
  endif()
 endforeach()
 add_custom_target(color_shaders ALL DEPENDS ${COLOR_SPIRV_OUTPUTS})
endif()
if(ANDROID)
 add_library(own_color_jni SHARED bridge/android_vulkan_jni.cpp)
 target_link_libraries(own_color_jni PRIVATE color::engine)
endif()
if(APPLE)
 enable_language(OBJCXX)
 add_library(own_color_apple STATIC bridge/ios_metal_bridge.mm)
 target_compile_options(own_color_apple PRIVATE -fobjc-arc)
 target_link_libraries(own_color_apple PRIVATE color::engine "-framework Foundation" "-framework Metal")
endif()
option(COLOR_BUILD_VULKAN "Build original POSIX/Android Vulkan compute dispatcher" OFF)
set(COLOR_VULKAN_HEADERS "" CACHE PATH "Vulkan-Headers include directory")
if(COLOR_BUILD_VULKAN)
 find_path(COLOR_VULKAN_INCLUDE vulkan/vulkan.h HINTS "${COLOR_VULKAN_HEADERS}" REQUIRED)
 add_library(own_color_vulkan src/vulkan_compute.cpp)
 target_include_directories(own_color_vulkan PRIVATE "${COLOR_VULKAN_INCLUDE}")
 target_link_libraries(own_color_vulkan PUBLIC color::engine PRIVATE ${CMAKE_DL_LIBS})
 if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(own_color_vulkan PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
 endif()
endif()
if(COLOR_BUILD_VULKAN AND COLOR_BUILD_TESTS AND COLOR_GLSLANG)
 add_executable(color_vulkan_test tests/vulkan_test.cpp)
 target_link_libraries(color_vulkan_test PRIVATE own_color_vulkan)
 target_compile_definitions(color_vulkan_test PRIVATE COLOR_GRADE_SPIRV="${CMAKE_CURRENT_BINARY_DIR}/spirv/grade.comp.spv")
 add_dependencies(color_vulkan_test color_shaders)
 add_test(NAME color_vulkan_readback COMMAND color_vulkan_test)
 set_tests_properties(color_vulkan_readback PROPERTIES SKIP_RETURN_CODE 77)
endif()
if(COLOR_BUILD_VULKAN AND COLOR_BUILD_TESTS AND COLOR_GLSLANG)
 add_executable(color_gpu_parity tests/gpu_parity.cpp)
 target_link_libraries(color_gpu_parity PRIVATE own_color_vulkan)
 target_compile_definitions(color_gpu_parity PRIVATE COLOR_SPIRV_DIRECTORY="${CMAKE_CURRENT_BINARY_DIR}/spirv")
 add_dependencies(color_gpu_parity color_shaders)
 add_test(NAME color_gpu_operator_parity COMMAND color_gpu_parity)
 set_tests_properties(color_gpu_operator_parity PROPERTIES SKIP_RETURN_CODE 77)
endif()
